    37→    37→// the type icon shown in the column header
    38→    38→function typeIcon(t: CellType) {
    39→    39→  const c = "w-3 h-3 text-muted-foreground";
    40→    40→  switch (t) {
    41→    41→    case "number":
    42→    42→      return <Hash className={c} />;
    43→    43→    case "date":
    44→    44→      return <Calendar className={c} />;
    45→    45→    case "link":
    46→    46→      return <Link2 className={c} />;
    47→    47→    case "image":
    48→    48→      return <ImageIcon className={c} />;
    49→    49→    case "audio":
    50→    50→      return <Music className={c} />;
    51→    51→    case "file":
    52→    52→      return <FileText className={c} />;
    53→    53→    case "longtext":
    54→    54→      return <AlignLeft className={c} />;
    55→    55→    case "markdown":
    56→    56→      return <FileText className={c} />;
    57→    57→    default:
    58→    58→      return <Type className={c} />;
    59→    59→  }
    60→    60→}

    62→    62→interface GridProps {
    63→    63→  data: GridData;
    64→    64→  onChange: (next: GridData) => void;
    65→    65→  // depth of nesting (0 = top). affects styling only.
    66→    66→  depth?: number;

    68→    68→
    69→    69→export function Grid({ data, onChange, depth = 0 }: GridProps) {
    70→    70→  const { columns, rows } = data;
    71→    71→
    72→    72→  // ---- column operations ----
    73→    73→  const updateColumn = (id: string, patch: Partial<ColumnDef>) =>
    74→    74→    onChange({
    75→    75→      ...data,
    76→    76→      columns: columns.map((c) => (c.id === id ? { ...c, ...patch } : c)),
    77→    77→    });
    78→    78→
    79→    79→  const addColumn = () => {
    80→    80→    const idx = columns.length;
    81→    81→    const col = makeColumn(`field ${idx + 1}`, "text", 180);
    82→    82→    onChange({
    83→    83→      ...data,
    84→    84→      columns: [...columns, col],
    85→    85→      rows: rows.map((r) => ({ ...r, cells: [...r.cells, makeCell()] })),
    86→    86→    });
    87→    87→  };
    88→    88→
    89→    89→  const removeColumn = (id: string) => {
    90→    90→    const i = columns.findIndex((c) => c.id === id);
    91→    91→    if (i < 0) return;
    92→    92→    onChange({
    93→    93→      ...data,
    94→    94→      columns: columns.filter((c) => c.id !== id),
    95→    95→      rows: rows.map((r) => ({
    96→    96→        ...r,
    97→    97→        cells: r.cells.filter((_, ci) => ci !== i),
    98→    98→      })),
    99→    99→    });
   100→   100→  };
   101→   101→
   102→   102→  const reorderColumn = (from: number, to: number) => {
   103→   103→    if (from === to || from < 0 || to < 0) return;
   104→   104→    const cols = [...columns];
   105→   105→    const [moved] = cols.splice(from, 1);
   106→   106→    cols.splice(to, 0, moved);
   107→   107→    const newRows = rows.map((r) => {
   108→   108→      const cells = [...r.cells];
   109→   109→      const [mc] = cells.splice(from, 1);
   110→   110→      cells.splice(to, 0, mc);
   111→   111→      return { ...r, cells };
   112→   112→    });
   113→   113→    onChange({ ...data, columns: cols, rows: newRows });
   114→   114→  };
   115→   115→
   116→   116→  // ---- row operations ----
   117→   117→  const updateRow = (id: string, patch: Partial<RowDef>) =>
   118→   118→    onChange({
   119→   119→      ...data,
   120→   120→      rows: rows.map((r) => (r.id === id ? { ...r, ...patch } : r)),
   121→   121→    });
   122→   122→
   123→   123→  const addRow = () => {
   124→   124→    const row = makeRow(columns.length, 40);
   125→   125→    onChange({ ...data, rows: [...rows, row] });
   126→   126→  };
   127→   127→
   128→   128→  const removeRow = (id: string) =>
   129→   129→    onChange({ ...data, rows: rows.filter((r) => r.id !== id) });
   130→   130→
   131→   131→  const reorderRow = (from: number, to: number) => {
   132→   132→    if (from === to || from < 0 || to < 0) return;
   133→   133→    const r = [...rows];
   134→   134→    const [moved] = r.splice(from, 1);
   135→   135→    r.splice(to, 0, moved);
   136→   136→    onChange({ ...data, rows: r });
   137→   137→  };
   138→   138→
   139→   139→  // ---- cell operations ----
   140→   140→  const updateCell = (rowId: string, colId: string, patch: Partial<CellDef>) => {
   141→   141→    onChange({
   142→   142→      ...data,
   143→   143→      rows: rows.map((r) =>
   144→   144→        r.id === rowId
   145→   145→          ? {
   146→   146→              ...r,
   147→   147→              cells: r.cells.map((c, ci) => {
   148→   148→                if (columns[ci]?.id !== colId) return c;
   149→   149→                return { ...c, ...patch };
   150→   150→              }),
   151→   151→            }
   152→   152→          : r
   153→   153→      ),
   154→   154→    });
   155→   155→  };
   156→   156→
   157→   157→  const setCellValue = (rowId: string, colId: string, value: string | null) =>
   158→   158→    updateCell(rowId, colId, { value });
   159→   159→
   160→   160→  const setCellNested = (rowId: string, colId: string, nestedGrid: GridData | null) =>
   161→   161→    updateCell(rowId, colId, { nestedGrid });
   162→   162→
   163→   163→  // total width for the sizing wrapper
   164→   164→  const rowHandleWidth = 34;
   165→   165→  const addColBtnWidth = 28;
   166→   166→  const totalWidth =
   167→   167→    rowHandleWidth +
   168→   168→    columns.reduce((s, c) => s + c.width, 0) +
   169→   169→    addColBtnWidth;
   170→   170→
   171→   171→  return (
   172→   172→    <div className="inline-block min-w-full" style={{ width: totalWidth }}>
   173→   173→      <div className="flex items-stretch border-b">
   174→   174→        {/* corner */}
   175→   175→        <div
   176→   176→          className="shrink-0 border-r bg-muted/40 grid place-items-center"
   177→   177→          style={{ width: rowHandleWidth }}
   178→   178→        >
   179→   179→          <span className="text-[9px] text-muted-foreground">{rows.length}</span>
   180→   180→        </div>
   181→   181→        {/* column headers */}
   182→   182→        {columns.map((col, i) => (
   183→   183→          <ColumnHeader
   184→   184→            key={col.id}
   185→   185→            col={col}
   186→   186→            index={i}
   187→   187→            onRename={(name) => updateColumn(col.id, { name })}
   188→   188→            onType={(type) => updateColumn(col.id, { type })}
   189→   189→            onWidth={(width) => updateColumn(col.id, { width })}
   190→   190→            onRemove={() => removeColumn(col.id)}
   191→   191→            onReorder={reorderColumn}
   192→   192→          />
   193→   193→        ))}
   194→   194→        {/* add column */}
   195→   195→        <div className="shrink-0 flex" style={{ width: addColBtnWidth }}>
   196→   196→          <button
   197→   197→            onClick={addColumn}
   198→   198→            className="w-full grid place-items-center btn-quiet text-muted-foreground"
   199→   199→          >
   200→   200→            <Plus className="w-3 h-3" />
   201→   201→          </button>
   202→   202→        </div>
   203→   203→      </div>
   204→   204→
   205→   205→      {/* rows */}
   206→   206→      <div>
   207→   207→        {rows.map((row, ri) => (
   208→   208→          <RowView
   209→   209→            key={row.id}
   210→   210→            row={row}
   211→   211→            index={ri}
   212→   212→            columns={columns}
   213→   213→            rowHandleWidth={rowHandleWidth}
   214→   214→            onHeight={(height) => updateRow(row.id, { height })}
   215→   215→            onRemove={() => removeRow(row.id)}
   216→   216→            onReorder={reorderRow}
   217→   217→            onCellValue={(colId, value) => setCellValue(row.id, colId, value)}
   218→   218→            onCellAttachment={(colId, attachmentId, filename) =>
   219→   219→              updateCell(row.id, colId, {
   220→   220→                attachmentId,
   221→   221→                value: filename ?? null,
   222→   222→              })
   223→   223→            }
   224→   224→            onCellClearAttachment={(colId) =>
   225→   225→              updateCell(row.id, colId, { attachmentId: null, value: null })
   226→   226→            }
   227→   227→            onCellNested={(colId, nestedGrid) =>
   228→   228→              setCellNested(row.id, colId, nestedGrid)
   229→   229→            }
   230→   230→          />
   231→   231→        ))}
   232→   232→        {rows.length === 0 && (
   233→   233→          <div className="h-16 grid place-items-center text-[10px] text-muted-foreground">
   234→   234→            empty
   235→   235→          </div>
   236→   236→        )}
   237→   237→      </div>
   238→   238→
   239→   239→      {/* add row */}
   240→   240→      <button
   241→   241→        onClick={addRow}
   242→   242→        className="h-8 w-full flex items-center gap-1.5 px-2 text-[10px] text-muted-foreground btn-quiet border-b"
   243→   243→      >
   244→   244→        <Plus className="w-3 h-3" />
   245→   245→        add row
   246→   246→      </button>
   247→   247→    </div>
   248→   248→  );
   249→   249→}
   250→   250→
   251→   251→// ---- column header ----
   252→   252→function ColumnHeader({
   253→   253→  col,
   254→   254→  index,
   255→   255→  onRename,
   256→   256→  onType,
   257→   257→  onWidth,
   258→   258→  onRemove,
   259→   259→  onReorder,
   260→   260→}: {
   261→   261→  col: ColumnDef;
   262→   262→  index: number;
   263→   263→  onRename: (name: string) => void;
   264→   264→  onType: (type: CellType) => void;
   265→   265→  onWidth: (width: number) => void;
   266→   266→  onRemove: () => void;
   267→   267→  onReorder: (from: number, to: number) => void;
   268→   268→}) {
   269→   269→  const [editing, setEditing] = useState(false);
   270→   270→  const [val, setVal] = useState(col.name);
   271→   271→  const [menuOpen, setMenuOpen] = useState(false);
   272→   272→  const [resizing, setResizing] = useState(false);
   273→   273→  const [dragOver, setDragOver] = useState(false);
   274→   274→  const menuRef = useRef<HTMLDivElement>(null);
   275→   275→
   276→   276→  useEffect(() => {
   277→   277→    if (!menuOpen) return;
   278→   278→    const handler = (e: MouseEvent) => {
   279→   279→      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
   280→   280→        setMenuOpen(false);
   281→   281→      }
   282→   282→    };
   283→   283→    document.addEventListener("mousedown", handler);
   284→   284→    return () => document.removeEventListener("mousedown", handler);
   285→   285→  }, [menuOpen]);
   286→   286→
   287→   287→  const startResize = (e: React.PointerEvent) => {
   288→   288→    e.preventDefault();
   289→   289→    e.stopPropagation();
   290→   290→    setResizing(true);
   291→   291→    const startX = e.clientX;
   292→   292→    const startWidth = col.width;
   293→   293→    const move = (ev: PointerEvent) => {
   294→   294→      const dx = ev.clientX - startX;
   295→   295→      onWidth(Math.max(60, startWidth + dx));
   296→   296→    };
   297→   297→    const up = () => {
   298→   298→      setResizing(false);
   299→   299→      window.removeEventListener("pointermove", move);
   300→   300→      window.removeEventListener("pointerup", up);
   301→   301→    };
   302→   302→    window.addEventListener("pointermove", move);
   303→   303→    window.addEventListener("pointerup", up);
   304→   304→  };
   305→   305→
   306→   306→  return (
   307→   307→    <div
   308→   308→      draggable={!editing && !resizing}
   309→   309→      onDragStart={(e) => {
   310→   310→        e.dataTransfer.setData("text/col", String(index));
   311→   311→        e.dataTransfer.effectAllowed = "move";
   312→   312→      }}
   313→   313→      onDragOver={(e) => {
   314→   314→        if (e.dataTransfer.types.includes("text/col")) {
   315→   315→          e.preventDefault();
   316→   316→          setDragOver(true);
   317→   317→        }
   318→   318→      }}
   319→   319→      onDragLeave={() => setDragOver(false)}
   320→   320→      onDrop={(e) => {
   321→   321→        e.preventDefault();
   322→   322→        setDragOver(false);
   323→   323→        const from = Number(e.dataTransfer.getData("text/col"));
   324→   324→        if (!Number.isNaN(from)) onReorder(from, index);
   325→   325→      }}
   326→   326→      className={cn(
   327→   327→        "relative shrink-0 border-r bg-muted/30 flex items-center group select-none",
   328→   328→        dragOver && "bg-accent"
   329→   329→      )}
   330→   330→      style={{ width: col.width }}
   331→   331→    >
   332→   332→      <button
   333→   333→        onClick={() => setMenuOpen((o) => !o)}
   334→   334→        className="h-7 px-1.5 flex items-center gap-1 btn-quiet shrink-0"
   335→   335→      >
   336→   336→        {typeIcon(col.type)}
   337→   337→      </button>
   338→   338→      <div className="flex-1 min-w-0">
   339→   339→        {editing ? (
   340→   340→          <input
   341→   341→            value={val}
   342→   342→            onChange={(e) => setVal(e.target.value)}
   343→   343→            onBlur={() => {
   344→   344→              onRename(val.trim() || "field");
   345→   345→              setEditing(false);
   346→   346→            }}
   347→   347→            onKeyDown={(e) => {
   348→   348→              if (e.key === "Enter") {
   349→   349→                onRename(val.trim() || "field");
   350→   350→                setEditing(false);
   351→   351→              }
   352→   352→              if (e.key === "Escape") {
   353→   353→                setVal(col.name);
   354→   354→                setEditing(false);
   355→   355→              }
   356→   356→            }}
   357→   357→            autoFocus
   358→   358→            className="w-full h-7 bg-transparent outline-none text-[11px] px-0.5"
   359→   359→          />
   360→   360→        ) : (
   361→   361→          <button
   362→   362→            onClick={() => {
   363→   363→              setVal(col.name);
   364→   364→              setEditing(true);
   365→   365→            }}
   366→   366→            className="w-full h-7 text-left text-[11px] px-0.5 truncate"
   367→   367→            title={col.name}
   368→   368→          >
   369→   369→            {col.name}
   370→   370→          </button>
   371→   371→        )}
   372→   372→      </div>
   373→   373→
   374→   374→      {/* type + actions menu */}
   375→   375→      {menuOpen && (
   376→   376→        <div
   377→   377→          ref={menuRef}
   378→   378→          className="absolute top-7 left-0 z-30 w-36 bg-popover border rounded-md shadow-sm py-1"
   379→   379→        >
   380→   380→          <div className="px-2 py-1 text-[9px] text-muted-foreground">type</div>
   381→   381→          {CELL_TYPES.map((t) => (
   382→   382→            <button
   383→   383→              key={t}
   384→   384→              onClick={() => {
   385→   385→                onType(t);
   386→   386→                setMenuOpen(false);
   387→   387→              }}
   388→   388→              className={cn(
   389→   389→                "w-full flex items-center gap-2 px-2 py-1.5 text-[11px] btn-quiet text-left",
   390→   390→                col.type === t && "font-medium"
   391→   391→              )}
   392→   392→            >
   393→   393→              {typeIcon(t)}
   394→   394→              {CELL_TYPE_LABELS[t]}
   395→   395→            </button>
   396→   396→          ))}
   397→   397→          <div className="h-px bg-border my-1" />
   398→   398→          <button
   399→   399→            onClick={() => {
   400→   400→              onRemove();
   401→   401→              setMenuOpen(false);
   402→   402→            }}
   403→   403→            className="w-full flex items-center gap-2 px-2 py-1.5 text-[11px] btn-quiet text-left"
   404→   404→          >
   405→   405→            <Trash2 className="w-3 h-3" />
   406→   406→            delete column
   407→   407→          </button>
   408→   408→        </div>
   409→   409→      )}
   410→   410→
   411→   411→      {/* resize handle */}
   412→   412→      <div
   413→   413→        onPointerDown={startResize}
   414→   414→        className="absolute top-0 right-0 w-1.5 h-full cursor-col-resize hover:bg-foreground/20"
   415→   415→      />
   416→   416→    </div>
   417→   417→  );
   418→   418→}

   420→   420→// ---- row ----
   421→   421→function RowView({
   422→   422→  row,
   423→   423→  index,
   424→   424→  columns,
   425→   425→  rowHandleWidth,
   426→   426→  onHeight,
   427→   427→  onRemove,
   428→   428→  onReorder,
   429→   429→  onCellValue,
   430→   430→  onCellAttachment,
   431→   431→  onCellClearAttachment,
   432→   432→  onCellNested,
   433→   433→}: {
   434→   434→  row: RowDef;
   435→   435→  index: number;
   436→   436→  columns: ColumnDef[];
   437→   437→  rowHandleWidth: number;
   438→   438→  onHeight: (h: number) => void;
   439→   439→  onRemove: () => void;
   440→   440→  onReorder: (from: number, to: number) => void;
   441→   441→  onCellValue: (colId: string, value: string | null) => void;
   442→   442→  onCellAttachment: (
   443→   443→    colId: string,
   444→   444→    attachmentId: string | null,
   445→   445→    filename: string | null
   446→   446→  ) => void;
   447→   447→  onCellClearAttachment: (colId: string) => void;
   448→   448→  onCellNested: (colId: string, nestedGrid: GridData | null) => void;
   449→   449→}) {
   450→   450→  const [resizing, setResizing] = useState(false);
   451→   451→  const [dragOver, setDragOver] = useState(false);
   452→   452→  const [menuOpen, setMenuOpen] = useState(false);
   453→   453→  const menuRef = useRef<HTMLDivElement>(null);
   454→   454→
   455→   455→  useEffect(() => {
   456→   456→    if (!menuOpen) return;
   457→   457→    const handler = (e: MouseEvent) => {
   458→   458→      if (menuRef.current && !menuRef.current.contains(e.target as Node))
   459→   459→        setMenuOpen(false);
   460→   460→    };
   461→   461→    document.addEventListener("mousedown", handler);
   462→   462→    return () => document.removeEventListener("mousedown", handler);
   463→   463→  }, [menuOpen]);
   464→   464→
   465→   465→  const startResize = (e: React.PointerEvent) => {
   466→   466→    e.preventDefault();
   467→   467→    e.stopPropagation();
   468→   468→    setResizing(true);
   469→   469→    const startY = e.clientY;
   470→   470→    const startH = row.height;
   471→   471→    const move = (ev: PointerEvent) =>
   472→   472→      onHeight(Math.max(28, startH + (ev.clientY - startY)));
   473→   473→    const up = () => {
   474→   474→      setResizing(false);
   475→   475→      window.removeEventListener("pointermove", move);
   476→   476→      window.removeEventListener("pointerup", up);
   477→   477→    };
   478→   478→    window.addEventListener("pointermove", move);
   479→   479→    window.addEventListener("pointerup", up);
   480→   480→  };
   481→   481→
   482→   482→  // a row may grow taller than its declared height if its cells contain
   483→   483→  // nested grids or media. we use min-height so content drives the height.
   484→   484→  return (
   485→   485→    <div
   486→   486→      draggable={!resizing}
   487→   487→      onDragStart={(e) => {
   488→   488→        e.dataTransfer.setData("text/row", String(index));
   489→   489→        e.dataTransfer.effectAllowed = "move";
   490→   490→      }}
   491→   491→      onDragOver={(e) => {
   492→   492→        if (e.dataTransfer.types.includes("text/row")) {
   493→   493→          e.preventDefault();
   494→   494→          setDragOver(true);
   495→   495→        }
   496→   496→      }}
   497→   497→      onDragLeave={() => setDragOver(false)}
   498→   498→      onDrop={(e) => {
   499→   499→        e.preventDefault();
   500→   500→        setDragOver(false);
   501→   501→        const from = Number(e.dataTransfer.getData("text/row"));
   502→   502→        if (!Number.isNaN(from)) onReorder(from, index);
   503→   503→      }}
   504→   504→      className={cn(

   506→   506→        dragOver && "bg-accent/60"
   507→   507→      )}
   508→   508→    >
   509→   509→      {/* row handle */}
   510→   510→      <div
   511→   511→        className="shrink-0 border-r bg-muted/30 flex flex-col items-center justify-center relative"
   512→   512→        style={{ width: rowHandleWidth, minHeight: row.height }}
   513→   513→      >

   516→   516→          className="h-5 w-5 grid place-items-center text-muted-foreground btn-quiet rounded"
   517→   517→        >
   518→   518→          <GripVertical className="w-3 h-3" />

   537→   537→      </div>

   539→   539→      {/* cells */}

   555→   555→
   556→   556→      {/* row resize handle */}
   557→   557→      <div

   559→   559→        className="absolute bottom-0 left-0 right-0 h-1 cursor-row-resize hover:bg-foreground/20"
   560→   560→      />
   561→   561→    </div>
   562→   562→  );
   563→   563→}

   864→   864→          </div>
   865→   865→        </div>
   866→   866→      )}
