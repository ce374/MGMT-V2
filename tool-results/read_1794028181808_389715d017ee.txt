    37→// the type icon shown in the column header
    38→function typeIcon(t: CellType) {
    39→  const c = "w-3 h-3 text-muted-foreground";
    40→  switch (t) {
    41→    case "number":
    42→      return <Hash className={c} />;
    43→    case "date":
    44→      return <Calendar className={c} />;
    45→    case "link":
    46→      return <Link2 className={c} />;
    47→    case "image":
    48→      return <ImageIcon className={c} />;
    49→    case "audio":
    50→      return <Music className={c} />;
    51→    case "file":
    52→      return <FileText className={c} />;
    53→    case "longtext":
    54→      return <AlignLeft className={c} />;
    55→    case "markdown":
    56→      return <FileText className={c} />;
    57→    default:
    58→      return <Type className={c} />;
    59→  }
    60→}

    62→interface GridProps {
    63→  data: GridData;
    64→  onChange: (next: GridData) => void;
    65→  // depth of nesting (0 = top). affects styling only.
    66→  depth?: number;
    67→}
    68→
    69→export function Grid({ data, onChange, depth = 0 }: GridProps) {
    70→  const { columns, rows } = data;
    71→
    72→  // ---- column operations ----
    73→  const updateColumn = (id: string, patch: Partial<ColumnDef>) =>
    74→    onChange({
    75→      ...data,
    76→      columns: columns.map((c) => (c.id === id ? { ...c, ...patch } : c)),
    77→    });
    78→
    79→  const addColumn = () => {
    80→    const idx = columns.length;
    81→    const col = makeColumn(`field ${idx + 1}`, "text", 180);
    82→    onChange({
    83→      ...data,
    84→      columns: [...columns, col],
    85→      rows: rows.map((r) => ({ ...r, cells: [...r.cells, makeCell()] })),
    86→    });
    87→  };
    88→
    89→  const removeColumn = (id: string) => {
    90→    const i = columns.findIndex((c) => c.id === id);
    91→    if (i < 0) return;
    92→    onChange({
    93→      ...data,
    94→      columns: columns.filter((c) => c.id !== id),
    95→      rows: rows.map((r) => ({
    96→        ...r,
    97→        cells: r.cells.filter((_, ci) => ci !== i),
    98→      })),
    99→    });
   100→  };
   101→
   102→  const reorderColumn = (from: number, to: number) => {
   103→    if (from === to || from < 0 || to < 0) return;
   104→    const cols = [...columns];
   105→    const [moved] = cols.splice(from, 1);
   106→    cols.splice(to, 0, moved);
   107→    const newRows = rows.map((r) => {
   108→      const cells = [...r.cells];
   109→      const [mc] = cells.splice(from, 1);
   110→      cells.splice(to, 0, mc);
   111→      return { ...r, cells };
   112→    });
   113→    onChange({ ...data, columns: cols, rows: newRows });
   114→  };
   115→
   116→  // ---- row operations ----
   117→  const updateRow = (id: string, patch: Partial<RowDef>) =>
   118→    onChange({
   119→      ...data,
   120→      rows: rows.map((r) => (r.id === id ? { ...r, ...patch } : r)),
   121→    });
   122→
   123→  const addRow = () => {
   124→    const row = makeRow(columns.length, 40);
   125→    onChange({ ...data, rows: [...rows, row] });
   126→  };
   127→
   128→  const removeRow = (id: string) =>
   129→    onChange({ ...data, rows: rows.filter((r) => r.id !== id) });
   130→
   131→  const reorderRow = (from: number, to: number) => {
   132→    if (from === to || from < 0 || to < 0) return;
   133→    const r = [...rows];
   134→    const [moved] = r.splice(from, 1);
   135→    r.splice(to, 0, moved);
   136→    onChange({ ...data, rows: r });
   137→  };
   138→
   139→  // ---- cell operations ----
   140→  const updateCell = (rowId: string, colId: string, patch: Partial<CellDef>) => {
   141→    onChange({
   142→      ...data,
   143→      rows: rows.map((r) =>
   144→        r.id === rowId
   145→          ? {
   146→              ...r,
   147→              cells: r.cells.map((c, ci) => {
   148→                if (columns[ci]?.id !== colId) return c;
   149→                return { ...c, ...patch };
   150→              }),
   151→            }
   152→          : r
   153→      ),
   154→    });
   155→  };
   156→
   157→  const setCellValue = (rowId: string, colId: string, value: string | null) =>
   158→    updateCell(rowId, colId, { value });
   159→
   160→  const setCellNested = (rowId: string, colId: string, nestedGrid: GridData | null) =>
   161→    updateCell(rowId, colId, { nestedGrid });
   162→
   163→  // total width for the sizing wrapper
   164→  const rowHandleWidth = 34;
   165→  const addColBtnWidth = 28;
   166→  const totalWidth =
   167→    rowHandleWidth +
   168→    columns.reduce((s, c) => s + c.width, 0) +
   169→    addColBtnWidth;
   170→
   171→  return (
   172→    <div className="inline-block min-w-full" style={{ width: totalWidth }}>
   173→      <div className="flex items-stretch border-b">
   174→        {/* corner */}
   175→        <div
   176→          className="shrink-0 border-r bg-muted/40 grid place-items-center"
   177→          style={{ width: rowHandleWidth }}
   178→        >
   179→          <span className="text-[9px] text-muted-foreground">{rows.length}</span>
   180→        </div>
   181→        {/* column headers */}
   182→        {columns.map((col, i) => (
   183→          <ColumnHeader
   184→            key={col.id}
   185→            col={col}
   186→            index={i}
   187→            onRename={(name) => updateColumn(col.id, { name })}
   188→            onType={(type) => updateColumn(col.id, { type })}
   189→            onWidth={(width) => updateColumn(col.id, { width })}
   190→            onRemove={() => removeColumn(col.id)}
   191→            onReorder={reorderColumn}
   192→          />
   193→        ))}
   194→        {/* add column */}
   195→        <div className="shrink-0 flex" style={{ width: addColBtnWidth }}>
   196→          <button
   197→            onClick={addColumn}
   198→            className="w-full grid place-items-center btn-quiet text-muted-foreground"
   199→          >
   200→            <Plus className="w-3 h-3" />
   201→          </button>
   202→        </div>
   203→      </div>
   204→
   205→      {/* rows */}
   206→      <div>
   207→        {rows.map((row, ri) => (
   208→          <RowView
   209→            key={row.id}
   210→            row={row}
   211→            index={ri}
   212→            columns={columns}
   213→            rowHandleWidth={rowHandleWidth}
   214→            onHeight={(height) => updateRow(row.id, { height })}
   215→            onRemove={() => removeRow(row.id)}
   216→            onReorder={reorderRow}
   217→            onCellValue={(colId, value) => setCellValue(row.id, colId, value)}
   218→            onCellAttachment={(colId, attachmentId, filename) =>
   219→              updateCell(row.id, colId, {
   220→                attachmentId,
   221→                value: filename ?? null,
   222→              })
   223→            }
   224→            onCellClearAttachment={(colId) =>
   225→              updateCell(row.id, colId, { attachmentId: null, value: null })
   226→            }
   227→            onCellNested={(colId, nestedGrid) =>
   228→              setCellNested(row.id, colId, nestedGrid)
   229→            }
   230→          />
   231→        ))}
   232→        {rows.length === 0 && (
   233→          <div className="h-16 grid place-items-center text-[10px] text-muted-foreground">
   234→            empty
   235→          </div>
   236→        )}
   237→      </div>
   238→
   239→      {/* add row */}
   240→      <button
   241→        onClick={addRow}
   242→        className="h-8 w-full flex items-center gap-1.5 px-2 text-[10px] text-muted-foreground btn-quiet border-b"
   243→      >
   244→        <Plus className="w-3 h-3" />
   245→        add row
   246→      </button>
   247→    </div>
   248→  );
   249→}

   251→// ---- column header ----
   252→function ColumnHeader({
   253→  col,
   254→  index,
   255→  onRename,
   256→  onType,
   257→  onWidth,
   258→  onRemove,
   259→  onReorder,
   260→}: {
   261→  col: ColumnDef;
   262→  index: number;
   263→  onRename: (name: string) => void;
   264→  onType: (type: CellType) => void;
   265→  onWidth: (width: number) => void;
   266→  onRemove: () => void;
   267→  onReorder: (from: number, to: number) => void;
   268→}) {
   269→  const [editing, setEditing] = useState(false);
   270→  const [val, setVal] = useState(col.name);
   271→  const [menuOpen, setMenuOpen] = useState(false);
   272→  const [resizing, setResizing] = useState(false);
   273→  const [dragOver, setDragOver] = useState(false);
   274→  const menuRef = useRef<HTMLDivElement>(null);
   275→
   276→  useEffect(() => {
   277→    if (!menuOpen) return;
   278→    const handler = (e: MouseEvent) => {
   279→      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
   280→        setMenuOpen(false);
   281→      }
   282→    };
   283→    document.addEventListener("mousedown", handler);
   284→    return () => document.removeEventListener("mousedown", handler);
   285→  }, [menuOpen]);
   286→
   287→  const startResize = (e: React.PointerEvent) => {
   288→    e.preventDefault();
   289→    e.stopPropagation();
   290→    setResizing(true);
   291→    const startX = e.clientX;
   292→    const startWidth = col.width;
   293→    const move = (ev: PointerEvent) => {
   294→      const dx = ev.clientX - startX;
   295→      onWidth(Math.max(60, startWidth + dx));
   296→    };
   297→    const up = () => {
   298→      setResizing(false);
   299→      window.removeEventListener("pointermove", move);
   300→      window.removeEventListener("pointerup", up);
   301→    };
   302→    window.addEventListener("pointermove", move);
   303→    window.addEventListener("pointerup", up);
   304→  };
   305→
   306→  return (
   307→    <div
   308→      draggable={!editing && !resizing}
   309→      onDragStart={(e) => {
   310→        e.dataTransfer.setData("text/col", String(index));
   311→        e.dataTransfer.effectAllowed = "move";
   312→      }}
   313→      onDragOver={(e) => {
   314→        if (e.dataTransfer.types.includes("text/col")) {
   315→          e.preventDefault();
   316→          setDragOver(true);
   317→        }
   318→      }}
   319→      onDragLeave={() => setDragOver(false)}
   320→      onDrop={(e) => {
   321→        e.preventDefault();
   322→        setDragOver(false);
   323→        const from = Number(e.dataTransfer.getData("text/col"));
   324→        if (!Number.isNaN(from)) onReorder(from, index);
   325→      }}
   326→      className={cn(
   327→        "relative shrink-0 border-r bg-muted/30 flex items-center group select-none",
   328→        dragOver && "bg-accent"
   329→      )}
   330→      style={{ width: col.width }}
   331→    >
   332→      <button
   333→        onClick={() => setMenuOpen((o) => !o)}
   334→        className="h-7 px-1.5 flex items-center gap-1 btn-quiet shrink-0"
   335→      >
   336→        {typeIcon(col.type)}
   337→      </button>
   338→      <div className="flex-1 min-w-0">
   339→        {editing ? (
   340→          <input
   341→            value={val}
   342→            onChange={(e) => setVal(e.target.value)}
   343→            onBlur={() => {
   344→              onRename(val.trim() || "field");
   345→              setEditing(false);
   346→            }}
   347→            onKeyDown={(e) => {
   348→              if (e.key === "Enter") {
   349→                onRename(val.trim() || "field");
   350→                setEditing(false);
   351→              }
   352→              if (e.key === "Escape") {
   353→                setVal(col.name);
   354→                setEditing(false);
   355→              }
   356→            }}
   357→            autoFocus
   358→            className="w-full h-7 bg-transparent outline-none text-[11px] px-0.5"
   359→          />
   360→        ) : (
   361→          <button
   362→            onClick={() => {
   363→              setVal(col.name);
   364→              setEditing(true);
   365→            }}
   366→            className="w-full h-7 text-left text-[11px] px-0.5 truncate"
   367→            title={col.name}
   368→          >
   369→            {col.name}
   370→          </button>
   371→        )}
   372→      </div>
   373→
   374→      {/* type + actions menu */}
   375→      {menuOpen && (
   376→        <div
   377→          ref={menuRef}
   378→          className="absolute top-7 left-0 z-30 w-36 bg-popover border rounded-md shadow-sm py-1"
   379→        >
   380→          <div className="px-2 py-1 text-[9px] text-muted-foreground">type</div>
   381→          {CELL_TYPES.map((t) => (
   382→            <button
   383→              key={t}
   384→              onClick={() => {
   385→                onType(t);
   386→                setMenuOpen(false);
   387→              }}
   388→              className={cn(
   389→                "w-full flex items-center gap-2 px-2 py-1.5 text-[11px] btn-quiet text-left",
   390→                col.type === t && "font-medium"
   391→              )}
   392→            >
   393→              {typeIcon(t)}
   394→              {CELL_TYPE_LABELS[t]}
   395→            </button>
   396→          ))}
   397→          <div className="h-px bg-border my-1" />
   398→          <button
   399→            onClick={() => {
   400→              onRemove();
   401→              setMenuOpen(false);
   402→            }}
   403→            className="w-full flex items-center gap-2 px-2 py-1.5 text-[11px] btn-quiet text-left"
   404→          >
   405→            <Trash2 className="w-3 h-3" />
   406→            delete column
   407→          </button>
   408→        </div>
   409→      )}
   410→
   411→      {/* resize handle */}
   412→      <div
   413→        onPointerDown={startResize}
   414→        className="absolute top-0 right-0 w-1.5 h-full cursor-col-resize hover:bg-foreground/20"
   415→      />
   416→    </div>
   417→  );
   418→}

   420→// ---- row ----
   421→function RowView({
   422→  row,
   423→  index,
   424→  columns,
   425→  rowHandleWidth,
   426→  onHeight,
   427→  onRemove,
   428→  onReorder,
   429→  onCellValue,
   430→  onCellAttachment,
   431→  onCellClearAttachment,
   432→  onCellNested,
   433→}: {
   434→  row: RowDef;
   435→  index: number;
   436→  columns: ColumnDef[];
   437→  rowHandleWidth: number;
   438→  onHeight: (h: number) => void;
   439→  onRemove: () => void;
   440→  onReorder: (from: number, to: number) => void;
   441→  onCellValue: (colId: string, value: string | null) => void;
   442→  onCellAttachment: (
   443→    colId: string,
   444→    attachmentId: string | null,
   445→    filename: string | null
   446→  ) => void;
   447→  onCellClearAttachment: (colId: string) => void;
   448→  onCellNested: (colId: string, nestedGrid: GridData | null) => void;
   449→}) {
   450→  const [resizing, setResizing] = useState(false);
   451→  const [dragOver, setDragOver] = useState(false);
   452→  const [menuOpen, setMenuOpen] = useState(false);
   453→  const menuRef = useRef<HTMLDivElement>(null);
   454→
   455→  useEffect(() => {
   456→    if (!menuOpen) return;
   457→    const handler = (e: MouseEvent) => {
   458→      if (menuRef.current && !menuRef.current.contains(e.target as Node))
   459→        setMenuOpen(false);
   460→    };
   461→    document.addEventListener("mousedown", handler);
   462→    return () => document.removeEventListener("mousedown", handler);
   463→  }, [menuOpen]);
   464→
   465→  const startResize = (e: React.PointerEvent) => {
   466→    e.preventDefault();
   467→    e.stopPropagation();
   468→    setResizing(true);
   469→    const startY = e.clientY;
   470→    const startH = row.height;
   471→    const move = (ev: PointerEvent) =>
   472→      onHeight(Math.max(28, startH + (ev.clientY - startY)));
   473→    const up = () => {
   474→      setResizing(false);
   475→      window.removeEventListener("pointermove", move);
   476→      window.removeEventListener("pointerup", up);
   477→    };
   478→    window.addEventListener("pointermove", move);
   479→    window.addEventListener("pointerup", up);
   480→  };
   481→
   482→  // a row may grow taller than its declared height if its cells contain
   483→  // nested grids or media. we use min-height so content drives the height.
   484→  return (
   485→    <div
   486→      draggable={!resizing}
   487→      onDragStart={(e) => {
   488→        e.dataTransfer.setData("text/row", String(index));
   489→        e.dataTransfer.effectAllowed = "move";
   490→      }}
   491→      onDragOver={(e) => {
   492→        if (e.dataTransfer.types.includes("text/row")) {
   493→          e.preventDefault();
   494→          setDragOver(true);
   495→        }
   496→      }}
   497→      onDragLeave={() => setDragOver(false)}
   498→      onDrop={(e) => {
   499→        e.preventDefault();
   500→        setDragOver(false);
   501→        const from = Number(e.dataTransfer.getData("text/row"));
   502→        if (!Number.isNaN(from)) onReorder(from, index);
   503→      }}
   504→      className={cn(
   505→        "flex items-stretch border-b relative group",
   506→        dragOver && "bg-accent/60"
   507→      )}
   508→    >
   509→      {/* row handle */}
   510→      <div
   511→        className="shrink-0 border-r bg-muted/30 flex flex-col items-center justify-center relative"
   512→        style={{ width: rowHandleWidth, minHeight: row.height }}
   513→      >
   514→        <button
   515→          onClick={() => setMenuOpen((o) => !o)}
   516→          className="h-5 w-5 grid place-items-center text-muted-foreground btn-quiet rounded"
   517→        >
   518→          <GripVertical className="w-3 h-3" />
   519→        </button>
   520→        {menuOpen && (
   521→          <div
   522→            ref={menuRef}
   523→            className="absolute top-5 left-0 z-30 w-28 bg-popover border rounded-md shadow-sm py-1"
   524→          >
   525→            <button
   526→              onClick={() => {
   527→                onRemove();
   528→                setMenuOpen(false);
   529→              }}
   530→              className="w-full flex items-center gap-2 px-2 py-1.5 text-[11px] btn-quiet text-left"
   531→            >
   532→              <Trash2 className="w-3 h-3" />
   533→              delete row
   534→            </button>
   535→          </div>
   536→        )}
   537→      </div>
   538→
   539→      {/* cells */}
   540→      {columns.map((col, ci) => {
   541→        const cell = row.cells[ci];
   542→        return (
   543→          <CellView
   544→            key={cell?.id ?? ci}
   545→            cell={cell}
   546→            col={col}
   547→            onValue={(v) => onCellValue(col.id, v)}
   548→            onAttachment={(aid, fn) => onCellAttachment(col.id, aid, fn)}
   549→            onClearAttachment={() => onCellClearAttachment(col.id)}
   550→            onNested={(ng) => onCellNested(col.id, ng)}
   551→          />
   552→        );
   553→      })}
   554→      <div className="flex-1 border-l-0" />
   555→
   556→      {/* row resize handle */}
   557→      <div
   558→        onPointerDown={startResize}
   559→        className="absolute bottom-0 left-0 right-0 h-1 cursor-row-resize hover:bg-foreground/20"
   560→      />
   561→    </div>
   562→  );
   563→}

   565→// ---- cell ----
   566→function CellView({
   567→  cell,
   568→  col,
   569→  onValue,
   570→  onAttachment,
   571→  onClearAttachment,
   572→  onNested,
   573→}: {
   574→  cell: CellDef | undefined;
   575→  col: ColumnDef;
   576→  onValue: (v: string | null) => void;
   577→  onAttachment: (attachmentId: string | null, filename: string | null) => void;
   578→  onClearAttachment: () => void;
   579→  onNested: (nestedGrid: GridData | null) => void;
   580→}) {
   581→  // ensure a cell always exists even if row.cells is short
   582→  const c: CellDef = cell ?? makeCell();
   583→  const [editing, setEditing] = useState(false);
   584→  const [val, setVal] = useState(c.value ?? "");
   585→  const [nestedExpanded, setNestedExpanded] = useState<boolean>(
   586→    !!c.nestedGrid
   587→  );
   588→  const fileRef = useRef<HTMLInputElement>(null);
   589→
   590→  const enterEdit = () => {
   591→    setVal(c.value ?? "");
   592→    setEditing(true);
   593→  };
   594→
   595→  const hasNested = !!c.nestedGrid;
   596→
   597→  const handleUpload = async (file: File) => {
   598→    try {
   599→      const att = await api.uploadAttachment(file);
   600→      onAttachment(att.id, att.filename);
   601→      toast.success("added");
   602→    } catch {
   603→      toast.error("upload failed");
   604→    }
   605→  };
   606→
   607→  const toggleNested = () => {
   608→    if (!c.nestedGrid) {
   609→      // create an empty nested grid with one column + one row
   610→      const ng: GridData = {
   611→        columns: [makeColumn("field 1", "text", 160)],
   612→        rows: [makeRow(1, 36)],
   613→      };
   614→      onNested(ng);
   615→      setNestedExpanded(true);
   616→    } else {
   617→      setNestedExpanded((e) => !e);
   618→    }
   619→  };
   620→
   621→  const renderContent = () => {
   622→    switch (col.type) {
   623→      case "image":
   624→        return c.attachmentId ? (
   625→          <ImageCell
   626→            attachmentId={c.attachmentId}
   627→            filename={c.value ?? ""}
   628→            onReplace={() => fileRef.current?.click()}
   629→            onRemove={() => onClearAttachment()}
   630→          />
   631→        ) : (
   632→          <EmptyMedia
   633→            icon={<ImageIcon className="w-3 h-3" />}
   634→            onClick={() => fileRef.current?.click()}
   635→          />
   636→        );
   637→      case "audio":
   638→        return c.attachmentId ? (
   639→          <AudioCell
   640→            attachmentId={c.attachmentId}
   641→            filename={c.value ?? ""}
   642→            onReplace={() => fileRef.current?.click()}
   643→            onRemove={() => onClearAttachment()}
   644→          />
   645→        ) : (
   646→          <EmptyMedia
   647→            icon={<Music className="w-3 h-3" />}
   648→            onClick={() => fileRef.current?.click()}
   649→          />
   650→        );
   651→      case "file":
   652→        return c.attachmentId ? (
   653→          <FileCell
   654→            attachmentId={c.attachmentId}
   655→            filename={c.value ?? ""}
   656→            onReplace={() => fileRef.current?.click()}
   657→            onRemove={() => onClearAttachment()}
   658→          />
   659→        ) : (
   660→          <EmptyMedia
   661→            icon={<FileText className="w-3 h-3" />}
   662→            onClick={() => fileRef.current?.click()}
   663→          />
   664→        );
   665→      case "markdown":
   666→        return (
   667→          <MarkdownCell
   668→            value={c.value}
   669→            onChange={(v) => onValue(v)}
   670→          />
   671→        );
   672→      case "longtext":
   673→        return editing ? (
   674→          <textarea
   675→            value={val}
   676→            onChange={(e) => setVal(e.target.value)}
   677→            onBlur={() => {
   678→              onValue(val);
   679→              setEditing(false);
   680→            }}
   681→            onKeyDown={(e) => {
   682→              if (e.key === "Escape") {
   683→                setVal(c.value ?? "");
   684→                setEditing(false);
   685→              }
   686→            }}
   687→            autoFocus
   688→            rows={4}
   689→            className="w-full h-full min-h-[80px] bg-transparent outline-none text-[11px] p-1.5 resize-y keep-case"
   690→          />
   691→        ) : (
   692→          <button
   693→            onClick={enterEdit}
   694→            className="w-full h-full text-left text-[11px] p-1.5 whitespace-pre-wrap break-words text-muted-foreground"
   695→          >
   696→            {c.value || "—"}
   697→          </button>
   698→        );
   699→      case "number":
   700→        return editing ? (
   701→          <input
   702→            value={val}
   703→            onChange={(e) => setVal(e.target.value)}
   704→            onBlur={() => {
   705→              onValue(val);
   706→              setEditing(false);
   707→            }}
   708→            onKeyDown={(e) => {
   709→              if (e.key === "Enter") {
   710→                onValue(val);
   711→                setEditing(false);
   712→              }
   713→              if (e.key === "Escape") setEditing(false);
   714→            }}
   715→            autoFocus
   716→            inputMode="decimal"
   717→            className="w-full h-full bg-transparent outline-none text-[11px] px-1.5 keep-case"
   718→          />
   719→        ) : (
   720→          <button
   721→            onClick={enterEdit}
   722→            className={cn(
   723→              "w-full h-full text-left text-[11px] px-1.5 truncate",
   724→              !c.value && "text-muted-foreground"
   725→            )}
   726→          >
   727→            {c.value || "—"}
   728→          </button>
   729→        );
   730→      case "date":
   731→        return (
   732→          <input
   733→            type="date"
   734→            value={c.value ?? ""}
   735→            onChange={(e) => onValue(e.target.value || null)}
   736→            className="w-full h-full bg-transparent outline-none text-[11px] px-1.5 keep-case"
   737→          />
   738→        );
   739→      case "link":
   740→        return c.value && !editing ? (
   741→          <a
   742→            href={c.value.startsWith("http") ? c.value : `https://${c.value}`}
   743→            target="_blank"
   744→            rel="noreferrer"
   745→            onClick={(e) => e.stopPropagation()}
   746→            className="w-full h-full flex items-center px-1.5 text-[11px] truncate underline underline-offset-2"
   747→          >
   748→            {c.value}
   749→          </a>
   750→        ) : editing ? (
   751→          <input
   752→            value={val}
   753→            onChange={(e) => setVal(e.target.value)}
   754→            onBlur={() => {
   755→              onValue(val);
   756→              setEditing(false);
   757→            }}
   758→            onKeyDown={(e) => {
   759→              if (e.key === "Enter") {
   760→                onValue(val);
   761→                setEditing(false);
   762→              }
   763→            }}
   764→            autoFocus
   765→            placeholder="https://"
   766→            className="w-full h-full bg-transparent outline-none text-[11px] px-1.5 keep-case"
   767→          />
   768→        ) : (
   769→          <button
   770→            onClick={enterEdit}
   771→            className={cn(
   772→              "w-full h-full text-left text-[11px] px-1.5 truncate",
   773→              !c.value && "text-muted-foreground"
   774→            )}
   775→          >
   776→            {c.value || "—"}
   777→          </button>
   778→        );
   779→      default: // text
   780→        return editing ? (
   781→          <input
   782→            value={val}
   783→            onChange={(e) => setVal(e.target.value)}
   784→            onBlur={() => {
   785→              onValue(val);
   786→              setEditing(false);
   787→            }}
   788→            onKeyDown={(e) => {
   789→              if (e.key === "Enter") {
   790→                onValue(val);
   791→                setEditing(false);
   792→              }
   793→              if (e.key === "Escape") {
   794→                setVal(c.value ?? "");
   795→                setEditing(false);
   796→              }
   797→            }}
   798→            autoFocus
   799→            className="w-full h-full bg-transparent outline-none text-[11px] px-1.5 keep-case"
   800→          />
   801→        ) : (
   802→          <button
   803→            onClick={enterEdit}
   804→            className={cn(
   805→              "w-full h-full text-left text-[11px] px-1.5 truncate",
   806→              !c.value && "text-muted-foreground"
   807→            )}
   808→          >
   809→            {c.value || "—"}
   810→          </button>
   811→        );
   812→    }
   813→  };
   814→
   815→  return (
   816→    <div
   817→      className="relative border-r shrink-0 group/cell"
   818→      style={{ width: col.width, minHeight: 40 }}
   819→    >
   820→      <div className="h-full w-full">{renderContent()}</div>
   821→
   822→      {/* hidden file input for media uploads */}
   823→      <input
   824→        ref={fileRef}
   825→        type="file"
   826→        className="hidden"
   827→        onChange={(e) => {
   828→          const f = e.target.files?.[0];
   829→          if (f) handleUpload(f);
   830→          e.target.value = "";
   831→        }}
   832→      />
   833→
   834→      {/* nested grid toggle — bottom-right of every cell. always visible
   835→          but subtle so it works on touch devices (no hover). */}
   836→      <button
   837→        onClick={toggleNested}
   838→        className={cn(
   839→          "absolute bottom-0.5 right-0.5 h-4 w-4 grid place-items-center rounded-sm btn-quiet transition-opacity",
   840→          hasNested
   841→            ? "opacity-90 bg-background border"
   842→            : "opacity-30 hover:opacity-100"
   843→        )}
   844→      >
   845→        {nestedExpanded ? (
   846→          <ChevronDown className="w-2.5 h-2.5" />
   847→        ) : (
   848→          <ChevronRight className="w-2.5 h-2.5" />
   849→        )}
   850→      </button>
   851→
   852→      {/* nested grid panel */}
   853→      {hasNested && nestedExpanded && c.nestedGrid && (
   854→        <div className="border-t bg-background">
   855→          <div className="p-1.5">
   856→            <NestedGridPanel
   857→              grid={c.nestedGrid}
   858→              onChange={(ng) => onNested(ng)}
   859→              onRemove={() => {
   860→                onNested(null);
   861→                setNestedExpanded(false);
   862→              }}
   863→            />
   864→          </div>
   865→        </div>
   866→      )}
   867→    </div>
   868→  );
   869→}

   950→function FileCell(props: {
   951→  attachmentId: string;
   952→  filename: string;
   953→  onReplace: () => void;
   954→  onRemove: () => void;
   955→}) {
   956→  const isPdf = props.filename.toLowerCase().endsWith(".pdf");
   957→  const isMd = /\.(md|markdown|txt)$/i.test(props.filename);
   958→  const [preview, setPreview] = useState(false);
   959→  return (
   960→    <div className="w-full h-full min-h-[40px] p-1 group/m relative">
   961→      <button
   962→        onClick={() => (isPdf || isMd ? setPreview(true) : undefined)}
   963→        className="w-full flex items-center gap-1.5 px-1.5 h-7 rounded border btn-quiet"
   964→      >
   965→        <FileText className="w-3 h-3 shrink-0" />
   966→        <span className="text-[10px] truncate keep-case">{props.filename}</span>
   967→      </button>
   968→      <div className="absolute top-1 right-1 flex gap-0.5 opacity-0 group-hover/m:opacity-100 transition-opacity">
   969→        <button
   970→          onClick={props.onReplace}
   971→          className="h-5 w-5 grid place-items-center rounded bg-background/90 border btn-quiet"
   972→        >
   973→          <Plus className="w-2.5 h-2.5" />
   974→        </button>
   975→        <button
   976→          onClick={props.onRemove}
   977→          className="h-5 w-5 grid place-items-center rounded bg-background/90 border btn-quiet"
   978→        >
   979→          <Trash2 className="w-2.5 h-2.5" />
   980→        </button>
   981→      </div>
   982→      {preview && (
   983→        <FilePreview
   984→          url={api.attachmentUrl(props.attachmentId)}
   985→          filename={props.filename}
   986→          onClose={() => setPreview(false)}
   987→        />
   988→      )}
   989→    </div>
   990→  );
   991→}
